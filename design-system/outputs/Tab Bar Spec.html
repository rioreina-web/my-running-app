05 LAYOUT  ·  SAFE AREA

Where it sits on the device.

Hand the bar to SwiftUI through .safeAreaInset(edge: .bottom) — that's the only correct way to keep the home-indicator clearance honest across iPhone SE through 16 Pro Max.

Hit target
78 × 50pt per tab — comfortably above Apple's 44pt minimum.
Background
Color.drip.paper — opaque, no blur. The bar is part of the page, not a floating chrome.
06 IMPLEMENTATION

SwiftUI skeleton.

Drop into DripTabBar.swift. Replace the TabView in MainTabView with a ZStack + .safeAreaInset. Remove the UITabBarAppearance code in RunningLogApp.swift once you're done.

// DripTabBar.swift
import SwiftUI

enum DripTab: String, CaseIterable, Hashable {
  case log, train, trends, coach, plan

  var label: String {
    switch self {
    case .log:    "Log"
    case .train:  "Train"
    case .trends: "Trends"
    case .coach:  "Coach"
    case .plan:   "Plan"
    }
  }
}

struct DripTabBar: View {
  @Binding var selected: DripTab
  var badged: Set<DripTab> = []
  var disabled: Set<DripTab> = []

  var body: some View {
    VStack(spacing: 0) {
      Rectangle().fill(Color.drip.rule).frame(height: 1)
      HStack(spacing: 0) {
        ForEach(DripTab.allCases, id: \.self) { tab in
          DripTabItem(
            tab: tab,
            isActive: selected == tab,
            isBadged: badged.contains(tab),
            isDisabled: disabled.contains(tab)
          ) {
            selected = tab
          }
          .frame(maxWidth: .infinity)
        }
      }
      .padding(.top, 10)
      .padding(.bottom, 12)
    }
    .background(Color.drip.paper)
    .sensoryFeedback(.selection, trigger: selected)
  }
}

struct DripTabItem: View {
  let tab: DripTab
  let isActive: Bool
  let isBadged: Bool
  let isDisabled: Bool
  let action: () -> Void

  @State private var isPressed = false

  var body: some View {
    VStack(spacing: 7) {
      ZStack {
        Circle()
          .strokeBorder(
            isActive ? Color.drip.coral : Color.drip.inkTertiary,
            lineWidth: 1.5
          )
        if isActive {
          Circle().fill(Color.drip.coral)
        }
      }
      .frame(width: 6, height: 6)
      .scaleEffect(isPressed ? 0.65 : 1.0)

      Text(tab.label.uppercased())
        .font(.system(size: 10,
                      weight: isActive ? .semibold : .regular,
                      design: .monospaced))
        .tracking(1.2) // 0.12em × 10pt
        .foregroundStyle(isActive ? Color.drip.ink : Color.drip.inkSecondary)
        .overlay(alignment: .topTrailing) {
          if isBadged {
            Circle()
              .fill(Color.drip.coral)
              .frame(width: 6, height: 6)
              .overlay(Circle().stroke(Color.drip.paper, lineWidth: 2))
              .offset(x: 9, y: -14)
          }
        }
    }
    .scaleEffect(isPressed ? 0.97 : 1.0)
    .opacity(isDisabled ? 0.32 : 1.0)
    .contentShape(Rectangle())
    .allowsHitTesting(!isDisabled)
    .animation(.easeOut(duration: 0.15), value: isActive)
    .animation(.easeOut(duration: 0.08), value: isPressed)
    .onTapGesture { action() }
    .simultaneousGesture(
      DragGesture(minimumDistance: 0)
        .onChanged { _ in isPressed = true }
        .onEnded   { _ in isPressed = false }
    )
  }
}

Implement Color.drip.{paper,rule,ink,inkSecondary,inkTertiary,coral} as static properties on Color if you don't already — the values are in §02 above.

// MainTabView.swift — replace TabView with this shape
struct MainTabView: View {
  @State private var selectedTab: DripTab = .log

  var body: some View {
    Group {
      switch selectedTab {
      case .log:    LogScreen()
      case .train:  TrainingScreen()
      case .trends: TrendsScreen()
      case .coach:  CoachScreen()
      case .plan:   PlanScreen()
      }
    }
    .safeAreaInset(edge: .bottom) {
      DripTabBar(selected: $selectedTab)
    }
  }
}

A switch tears each tab down on change, which loses scroll position and view-model state. If that matters, instantiate all five views once and toggle .opacity / .allowsHitTesting in a ZStack instead. Heavier memory; preserves state.
